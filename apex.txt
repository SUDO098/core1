public class calci {

    // Addition
    public static Integer add(Integer a, Integer b) {
        return a + b;
    }

    // Subtraction
    public static Integer subtract(Integer a, Integer b) {
        return a - b;
    }

    // Multiplication
    public static Integer multiply(Integer a, Integer b) {
        return a * b;
    }

    // Division
    public static Decimal divide(Integer a, Integer b) {

        if(b == 0) {
            return 0;
        }

        return (Decimal)a / b;
    }
}


Execute Anonymous Code

System.debug('Addition = ' + calci.add(10,5));

System.debug('Subtraction = ' + calci.subtract(10,5));

System.debug('Multiplication = ' + calci.multiply(10,5));

System.debug('Division = ' + calci.divide(10,5));
